       
CREATE TABLE booking
(
  user_id      INT      NOT NULL,
  field_id     INT      NOT NULL,
  booking_time DATETIME NOT NULL,
  CONSTRAINT UQ_booking_time UNIQUE (booking_time),
  CONSTRAINT FK_fields_TO_booking
    FOREIGN KEY (field_id)
    REFERENCES fields (id),
    CONSTRAINT FK_users_TO_booking
    FOREIGN KEY (user_id)
    REFERENCES users (id)
);

CREATE TABLE comments
(
  comm_id   INT      NOT NULL AUTO_INCREMENT,
  content   VARCHAR(300)  NOT NULL,
  user_id   INT      NOT NULL,
  vid_id    INT      NOT NULL,
  comm_date DATETIME NOT NULL,
  PRIMARY KEY (comm_id),
  CONSTRAINT FK_users_TO_comments
    FOREIGN KEY (user_id)
    REFERENCES users (id),
    CONSTRAINT FK_removids_TO_comments
    FOREIGN KEY (vid_id)
    REFERENCES removids (vid_id)
);

CREATE TABLE do_exercise
(
  user_id INT NOT NULL,
  fit_id  INT NOT NULL,
  CONSTRAINT FK_users_TO_do_exercise
    FOREIGN KEY (user_id)
    REFERENCES users (id),
    CONSTRAINT FK_fit_TO_do_exercise
    FOREIGN KEY (fit_id)
    REFERENCES fit (id)
);

CREATE TABLE fields
(
  id         INT     NOT NULL AUTO_INCREMENT,
  field_name VARCHAR(100) NOT NULL,
  location   VARCHAR(100) NOT NULL,
  price      DECIMAL NOT NULL,
  PRIMARY KEY (id)
);

CREATE TABLE fit
(
  id         INT     NOT NULL AUTO_INCREMENT,
  fit_name   VARCHAR(100) NOT NULL,
  sets       INT     NOT NULL DEFAULT 1,
  fidvid_dir VARCHAR(500) NOT NULL,
  duration   INT     NOT NULL,
  PRIMARY KEY (id)
);

CREATE TABLE food_recipe
(
  id        INT NOT NULL AUTO_INCREMENT,
  calories  INT NULL    ,
  food_type VARCHAR(20)  NULL    ,
  PRIMARY KEY (id)
);

CREATE TABLE foodsys
(
  id       INT         NOT NULL AUTO_INCREMENT,
  sys_name VARCHAR(30) NOT NULL,
  min_bmi  DECIMAL     NOT NULL DEFAULT 0,
  max_bmi  DECIMAL     NOT NULL DEFAULT 40,
  PRIMARY KEY (id),CONSTRAINT UQ_sys_name UNIQUE (sys_name)
);

CREATE TABLE join_tourn
(
  user_id  INT NOT NULL,
  tourn_id INT NOT NULL,
  team     INT NULL,
  CONSTRAINT FK_users_TO_join_tourn
    FOREIGN KEY (user_id)
    REFERENCES users (id),
    CONSTRAINT FK_tournaments_TO_join_tourn
    FOREIGN KEY (tourn_id)
    REFERENCES tournaments (id)    
);

CREATE TABLE learned
(
  user_id  INT NOT NULL,
  skill_id INT NOT NULL,
  CONSTRAINT FK_users_TO_learned
    FOREIGN KEY (user_id)
    REFERENCES users (id),
    CONSTRAINT FK_skills_TO_learned
    FOREIGN KEY (skill_id)
    REFERENCES skills (id)
);

CREATE TABLE personal_img
(
  img_id  INT     NOT NULL AUTO_INCREMENT,
  user_d  INT     NOT NULL,
  img_dir VARCHAR(500) NOT NULL,
  PRIMARY KEY (img_id),
  CONSTRAINT FK_users_TO_personal_img
    FOREIGN KEY (user_d)
    REFERENCES users (id)
);

  CREATE TABLE recipe_sys
  (
    sys_id    INT NOT NULL,
    recipe_id INT NOT NULL,
    CONSTRAINT FK_foodsys_TO_recipe_sys
      FOREIGN KEY (sys_id)
      REFERENCES foodsys (id),
      CONSTRAINT FK_food_recipe_TO_recipe_sys
      FOREIGN KEY (recipe_id)
      REFERENCES food_recipe (id)
  );

CREATE TABLE removids
(
  vid_id   INT      NOT NULL AUTO_INCREMENT,
  user_id  INT      NOT NULL,
  vid_dir  VARCHAR(500)  NOT NULL,
  caption  VARCHAR(300)  NULL    ,
  likes    INT      NOT NULL DEFAULT 0,
  pub_date DATETIME NOT NULL,
  PRIMARY KEY (vid_id),
  CONSTRAINT FK_users_TO_removids
    FOREIGN KEY (user_id)
    REFERENCES users (id)
);

CREATE TABLE skills
(
  id           INT         NOT NULL AUTO_INCREMENT,
  skill_name   VARCHAR(70) NOT NULL,
  point_gain   INT         NULL     DEFAULT 0,
  difficulty   VARCHAR(10) NULL    ,
  skillvid_dir VARCHAR(500)     NOT NULL,
  turvid_dir   VARCHAR(500)     NOT NULL,
  PRIMARY KEY (id)
);

CREATE TABLE tactics
(
  id           INT         NOT NULL AUTO_INCREMENT,
  tactics_name VARCHAR(20) NOT NULL,
  content      VARCHAR(65535)    NOT NULL,
  img_dir      VARCHAR(500)     NULL    ,
  PRIMARY KEY (id),
  CONSTRAINT UQ_tactics_name UNIQUE (tactics_name)
);

CREATE TABLE tournaments
(
  id          INT          NOT NULL AUTO_INCREMENT,
  tourn_name  VARCHAR(100) NOT NULL,
  governorate VARCHAR(30)  NOT NULL,
  rank        VARCHAR(15)  NOT NULL,
  start_date  DATETIME     NOT NULL,
  end_date    DATETIME     NOT NULL,
  fees        DECIMAL      NULL    ,
  field_id    INT          NOT NULL,
  teams_num   INT          NULL    ,
  PRIMARY KEY (id),
  CONSTRAINT FK_fields_TO_tournaments
    FOREIGN KEY (field_id)
    REFERENCES fields (id)
);

CREATE TABLE users
(
  id                 INT          NOT NULL AUTO_INCREMENT,
  name               VARCHAR(70)  NOT NULL,
  username           VARCHAR(70)  NOT NULL,
  email              VARCHAR(100) NOT NULL,
  password           VARCHAR(50)  NOT NULL COMMENT 'pass',
  weight             INT          NOT NULL COMMENT 'kg',
  height             INT          NOT NULL COMMENT 'cm',
  bdate              DATE         NOT NULL,
  country            VARCHAR(30)  NOT NULL,
  governorate        VARCHAR(30)  NOT NULL,
  football_position  VARCHAR(3)   NOT NULL,
  points             INT          NOT NULL DEFAULT 0,
  rank               VARCHAR(15)  NOT NULL DEFAULT beginners COMMENT 'depend on points',
  tournaments_played INT          NOT NULL DEFAULT 0,
  followers          INT          NOT NULL DEFAULT 0,
  bmi                DECIMAL      NOT NULL,
  PRIMARY KEY (id),
  CONSTRAINT UQ_username UNIQUE (username),
  CONSTRAINT UQ_email UNIQUE (email)
);
